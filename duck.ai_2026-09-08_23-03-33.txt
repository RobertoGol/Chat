```powershell
    param(
        [int]$X,
        [int]$Y,
        [string]$Text,

function Get-BoxLine {
    param(
        [int]$Width,
        [string]$Left = "│",
        [string]$Right = "│"
    )

    return "$Left" + ("─" * ($Width - 2)) + "$Right"
}

function Draw-Box {
    param(
        [int]$X,
        [int]$Y,
        [int]$Width,
        [int]$Height,
        [string]$Title,
        [string]$Color = $colors.Header
    )

}
```

```typescript
// src/gemini-browser.ts
import { chromium, Page } from "playwright";

export async function connectGemini(): Promise<Page> {
  const browser = await chromium.connectOverCDP(
    "http://127.0.0.1:9222"
  );

  for (const context of browser.contexts()) {
    for (const page of context.pages()) {
      if (page.url().includes("gemini.google.com")) {
        await page.bringToFront();
        return page;
      }
    }
  }

  throw new Error("Вкладка Gemini не найдена");
}

export async function sendToGemini(
  page: Page,
  message: string
): Promise<void> {
  const input = page.locator(
    'textarea, [contenteditable="true"]'
  ).last();

  await input.waitFor({ state: "visible", timeout: 15_000 });
  await input.fill(message);
  await input.press("Enter");
}

export async function readGeminiAnswer(
  page: Page
): Promise<string> {
  await page.waitForTimeout(1500);

  const candidates = page.locator(
    '[data-message-author-role="model"], model-response, message-content'
  );

  const count = await candidates.count();

  if (count === 0) {
    return "Ответ Gemini не найден.";
  }

  return (await candidates.last().innerText()).trim();
}
```
